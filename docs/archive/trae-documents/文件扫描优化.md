在 Go 语言中实现跨平台文件扫描，主要考虑以下几点：

## 1. 核心思路

```go
// 使用标准库的 filepath 包，它已处理不同系统的路径差异
import (
    "path/filepath"
    "os"
    "io/fs"
)
```

## 2. 基本实现方案

### 方案一：使用 filepath.Walk（简单但性能一般）
```go
func ScanWithWalk(root string) error {
    return filepath.Walk(root, func(path string, info os.FileInfo, err error) error {
        if err != nil {
            return err // 处理权限错误等
        }
        // 处理文件
        if !info.IsDir() {
            fmt.Printf("文件: %s, 大小: %d\n", path, info.Size())
        }
        return nil
    })
}
```

### 方案二：使用 filepath.WalkDir（Go 1.16+，推荐）
```go
func ScanWithWalkDir(root string) error {
    return filepath.WalkDir(root, func(path string, d fs.DirEntry, err error) error {
        if err != nil {
            // 可根据错误类型决定是否继续扫描
            if os.IsPermission(err) {
                return filepath.SkipDir // 跳过无权限目录
            }
            return err
        }
        
        // 过滤逻辑
        if d.IsDir() {
            // 跳过特定目录（如 .git, node_modules）
            if d.Name() == ".git" {
                return filepath.SkipDir
            }
            return nil
        }
        
        // 文件处理
        info, _ := d.Info()
        fmt.Printf("找到文件: %s\n", path)
        return nil
    })
}
```

### 方案三：并发扫描（高性能）
```go
type FileScanner struct {
    MaxWorkers int
    ExcludeDirs map[string]bool
    IncludeExt  []string
}

func (s *FileScanner) ConcurrentScan(root string) <-chan string {
    files := make(chan string, 1000)
    
    go func() {
        defer close(files)
        var wg sync.WaitGroup
        semaphore := make(chan struct{}, s.MaxWorkers)
        
        var scanDir func(string)
        scanDir = func(dir string) {
            defer wg.Done()
            
            entries, err := os.ReadDir(dir)
            if err != nil {
                return
            }
            
            for _, entry := range entries {
                fullPath := filepath.Join(dir, entry.Name())
                
                if entry.IsDir() {
                    // 排除目录检查
                    if s.ExcludeDirs[entry.Name()] {
                        continue
                    }
                    
                    wg.Add(1)
                    select {
                    case semaphore <- struct{}{}:
                        go func() {
                            scanDir(fullPath)
                            <-semaphore
                        }()
                    default:
                        scanDir(fullPath)
                    }
                } else {
                    // 文件扩展名过滤
                    if len(s.IncludeExt) > 0 {
                        ext := filepath.Ext(entry.Name())
                        matched := false
                        for _, e := range s.IncludeExt {
                            if strings.EqualFold(ext, e) {
                                matched = true
                                break
                            }
                        }
                        if !matched {
                            continue
                        }
                    }
                    files <- fullPath
                }
            }
        }
        
        wg.Add(1)
        scanDir(root)
        wg.Wait()
    }()
    
    return files
}
```

## 3. 跨平台注意事项

### 处理路径分隔符
```go
// 正确拼接路径
path := filepath.Join(dir, filename)

// 获取跨平台的工作目录
wd, _ := os.Getwd()

// 转换为系统路径格式
cleanPath := filepath.Clean(path)

// 分离目录和文件名
dir, file := filepath.Split(path)
```

### 处理系统差异
```go
func GetDefaultExclusions() []string {
    exclusions := []string{
        ".git", ".svn", ".hg", // 版本控制
        "node_modules", "__pycache__", // 编程语言相关
    }
    
    // 系统特定排除
    if runtime.GOOS == "darwin" {
        exclusions = append(exclusions, ".DS_Store")
    }
    if runtime.GOOS == "windows" {
        exclusions = append(exclusions, "Thumbs.db")
    }
    
    return exclusions
}
```

### 处理符号链接
```go
func isSymlink(path string) bool {
    if info, err := os.Lstat(path); err == nil {
        return info.Mode()&os.ModeSymlink != 0
    }
    return false
}

// 可选的符号链接处理
func SafeWalk(root string, followSymlinks bool) error {
    return filepath.Walk(root, func(path string, info os.FileInfo, err error) error {
        if err != nil {
            return err
        }
        
        if info.Mode()&os.ModeSymlink != 0 {
            if followSymlinks {
                // 解析符号链接
                target, err := os.Readlink(path)
                if err != nil {
                    return nil
                }
                // 注意避免循环链接
            }
            return nil // 或不处理符号链接
        }
        
        // 正常处理
        return nil
    })
}
```

## 4. 完整示例

```go
package main

import (
    "fmt"
    "io/fs"
    "os"
    "path/filepath"
    "runtime"
    "strings"
    "sync"
)

type ScannerConfig struct {
    RootDir      string
    MaxDepth     int
    FollowLinks  bool
    ExcludeDirs  []string
    IncludeExts  []string
    ExcludeExts  []string
    MinFileSize  int64
    MaxFileSize  int64
}

type FileScanner struct {
    config ScannerConfig
    mu     sync.RWMutex
}

func NewFileScanner(cfg ScannerConfig) *FileScanner {
    if cfg.MaxDepth <= 0 {
        cfg.MaxDepth = -1 // 无限深度
    }
    return &FileScanner{config: cfg}
}

func (s *FileScanner) Scan() ([]string, error) {
    var files []string
    var mu sync.Mutex
    
    err := s.walkDir(s.config.RootDir, 0, func(path string, info fs.FileInfo) error {
        // 大小过滤
        if s.config.MinFileSize > 0 && info.Size() < s.config.MinFileSize {
            return nil
        }
        if s.config.MaxFileSize > 0 && info.Size() > s.config.MaxFileSize {
            return nil
        }
        
        // 扩展名过滤
        ext := strings.ToLower(filepath.Ext(path))
        if len(s.config.IncludeExts) > 0 {
            found := false
            for _, e := range s.config.IncludeExts {
                if strings.EqualFold(ext, e) {
                    found = true
                    break
                }
            }
            if !found {
                return nil
            }
        }
        
        if len(s.config.ExcludeExts) > 0 {
            for _, e := range s.config.ExcludeExts {
                if strings.EqualFold(ext, e) {
                    return nil
                }
            }
        }
        
        mu.Lock()
        files = append(files, path)
        mu.Unlock()
        
        return nil
    })
    
    return files, err
}

func (s *FileScanner) walkDir(dir string, depth int, fn func(path string, info fs.FileInfo) error) error {
    if s.config.MaxDepth > 0 && depth > s.config.MaxDepth {
        return nil
    }
    
    entries, err := os.ReadDir(dir)
    if err != nil {
        if os.IsPermission(err) {
            return nil // 跳过无权限目录
        }
        return err
    }
    
    for _, entry := range entries {
        fullPath := filepath.Join(dir, entry.Name())
        
        // 排除目录检查
        if entry.IsDir() {
            skip := false
            for _, excl := range s.config.ExcludeDirs {
                if entry.Name() == excl {
                    skip = true
                    break
                }
            }
            if skip {
                continue
            }
            
            // 递归扫描子目录
            if err := s.walkDir(fullPath, depth+1, fn); err != nil {
                return err
            }
            continue
        }
        
        info, err := entry.Info()
        if err != nil {
            continue
        }
        
        // 处理符号链接
        if info.Mode()&os.ModeSymlink != 0 {
            if !s.config.FollowLinks {
                continue
            }
            // 注意：这里需要处理可能的循环链接
        }
        
        if err := fn(fullPath, info); err != nil {
            return err
        }
    }
    
    return nil
}

func main() {
    scanner := NewFileScanner(ScannerConfig{
        RootDir:     ".",
        MaxDepth:    5,
        FollowLinks: false,
        ExcludeDirs: []string{".git", "node_modules", "__pycache__"},
        IncludeExts: []string{".go", ".txt", ".md"},
        MinFileSize: 1024, // 至少1KB
    })
    
    files, err := scanner.Scan()
    if err != nil {
        fmt.Printf("扫描错误: %v\n", err)
        return
    }
    
    fmt.Printf("找到 %d 个文件\n", len(files))
    for _, file := range files {
        fmt.Println(file)
    }
}
```

## 5. 优化建议

1. **性能优化**：
   - 使用缓冲通道处理结果
   - 限制并发 goroutine 数量
   - 批量处理文件信息

2. **内存优化**：
   - 流式处理，避免一次性加载所有结果
   - 使用对象池复用数据结构

3. **错误处理**：
   - 区分权限错误和系统错误
   - 记录但继续扫描非致命错误

4. **平台适配**：
   - 使用 `runtime.GOOS` 判断操作系统
   - 处理不同系统的路径格式

这个方案考虑了不同操作系统的差异，提供了灵活的过滤选项，并支持并发扫描以提高性能。
